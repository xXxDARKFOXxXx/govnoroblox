local aliases = {
    body = "Body", lid = "Lid",
    wingl = "WingL", wingleft = "WingL",
    wingr = "WingR", wingright = "WingR",

local function groupName(object)
    if not (object:IsA("BasePart") or object:IsA("Model") or object:IsA("Folder")) then return nil end
    local key = string.lower(object.Name):gsub("[%s_%.%-]", ""):gsub("%d+$", "")
    if aliases[key] then return aliases[key] end
    for alias, name in pairs(aliases) do
        if key:sub(1, #alias + 4) == alias .. "mesh" then return name end
    end
    return nil
end
local groups = {Body = {}, Lid = {}, WingL = {}, WingR = {}}
local function discover(parent, inherited)
    for _, child in ipairs(parent:GetChildren()) do
        local name = groupName(child)
        if name and not inherited then table.insert(groups[name], child) end
        if name and inherited and name ~= inherited then
            error("Nested animation groups: " .. child:GetFullName() .. ". Body/Lid/WingL/WingR must be separate siblings")
        end
        discover(child, inherited or name)
    end
end

end

local function registerPart(part, group)
    if part == root then return end
    assert(not partGroups[part] or partGroups[part] == group, "Animation groups must not overlap")
    if partGroups[part] then return end
    partGroups[part] = group
    part.Anchored = true
    part.CanCollide = group == "Body"
    part.CanTouch = false
    table.insert(parts, {
        part = part, group = group,
        relative = originalPivot:ToObjectSpace(part.CFrame),
        transparency = part.Transparency,
        size = part.Size,
        color = part.Color,
        canCollide = part.CanCollide,
        canQuery = part.CanQuery,
    })
    if group == "Body" then table.insert(bodyParts, part) end
end
local function registerGroup(object, group)
    local count = 0
    if object:IsA("BasePart") then registerPart(object, group); count = count + 1 end
    for _, item in ipairs(object:GetDescendants()) do
        if item:IsA("BasePart") then registerPart(item, group); count = count + 1 end
    end
    assert(count > 0, group .. " contains no MeshParts/BaseParts")
end

local function pose(movingPivot, pulse, flap, lidOpen)
    local deltas = {}
    for _, name in ipairs({"WingL", "WingR"}) do
        local side = name == "WingL" and -1 or 1
        local hinge = hinges[name]
        deltas[name] = hinge * CFrame.Angles(0, 0, flap * side) * hinge:Inverse()
    end
    local hinge = hinges.Lid
    deltas.Lid = CFrame.new(0, CONFIG.LidLift * unit * lidOpen, 0) * hinge
        * CFrame.Angles(math.rad(-12) * lidOpen, math.rad(18) * lidOpen, 0) * hinge:Inverse()
    for _, data in ipairs(parts) do
        local relative = (deltas[data.group] or CFrame.identity) * data.relative
        local position = bodyCenter + (relative.Position - bodyCenter) * pulse
        data.part.Size = data.size * pulse
        data.part.CFrame = movingPivot * CFrame.new(position) * relative.Rotation
    end
    attachment.CFrame = root.CFrame:ToObjectSpace(movingPivot * CFrame.new(
        bodyCenter + Vector3.new(0, bodySize.Y * 0.15 * pulse, 0)))
end

local function particleEmitter(name, colorA, colorB, speed, size, lifetime, accel)
    local emitter = Instance.new("ParticleEmitter")
    emitter.Name = name
    emitter.Enabled = false
    emitter.Rate = 0
    emitter.Texture = "rbxasset://textures/particles/sparkles_main.dds"
    emitter.LightEmission = 1
    emitter.LightInfluence = 0
    emitter.Color = ColorSequence.new(colorA, colorB)
    emitter.Size = NumberSequence.new({
        NumberSequenceKeypoint.new(0, size * unit),
        NumberSequenceKeypoint.new(0.4, size * unit * 0.8),
        NumberSequenceKeypoint.new(1, 0),
    })
    emitter.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.02),
        NumberSequenceKeypoint.new(0.65, 0.22),
        NumberSequenceKeypoint.new(1, 1),
    })
    emitter.Lifetime = NumberRange.new(lifetime * 0.75, lifetime * 1.25)
    emitter.Speed = NumberRange.new(speed * unit * 0.55, speed * unit * 1.35)
    emitter.Drag = 1.6
    emitter.Acceleration = accel * unit
    emitter.RotSpeed = NumberRange.new(-170, 170)
    emitter.SpreadAngle = Vector2.new(180, 180)
    emitter.EmissionDirection = Enum.NormalId.Top
    emitter.Shape = Enum.ParticleEmitterShape.Sphere
    emitter.ShapeStyle = Enum.ParticleEmitterShapeStyle.Volume
    emitter.Parent = attachment
    return emitter
end

local tweens = {}
local function scaleTween(object, duration, props, style)
    local tween = TweenService:Create(object, TweenInfo.new(duration, style or Enum.EasingStyle.Quad,
        Enum.EasingDirection.Out), props)
    table.insert(tweens, tween)
    tween:Play()
    return tween
end
local function cancelTweens()
    for _, tween in ipairs(tweens) do tween:Cancel() end
    table.clear(tweens)
end

local function shockwave(at, color, duration, finalSize)

        Transparency = 1}, Enum.EasingStyle.Quart)
    Debris:AddItem(p, duration + 0.15)
end

local randomGen = Random.new()
local function giveReward(player)
    if not CONFIG.RewardsEnabled or player.Parent ~= Players then return end
    local rewards = ServerStorage:FindFirstChild("LuckyBlockRewards")
    if rewards then
        local tools = {}
        for _, item in ipairs(rewards:GetChildren()) do
            if item:IsA("Tool") then table.insert(tools, item) end
        end
        if #tools > 0 then
            local chosen = tools[randomGen:NextInteger(1, #tools)]
            local backpack = player:FindFirstChildOfClass("Backpack") or player:WaitForChild("Backpack", 4)
            if player.Parent ~= Players then return end
            if backpack then
                chosen:Clone().Parent = backpack
                notifyReward("🎁 " .. chosen.Name)
                return
            end
        end
    end
    local stats = player:FindFirstChild("leaderstats")
    if not stats then
        stats = Instance.new("Folder")
        stats.Name = "leaderstats"
        stats.Parent = player
    end
    local coins = stats:FindFirstChild("LuckyCoins")
    if not coins then
        coins = Instance.new("IntValue")
        coins.Name = "LuckyCoins"
        coins.Parent = stats
    end
    if not (coins:IsA("IntValue") or coins:IsA("NumberValue")) then
        warn("leaderstats.LuckyCoins should be IntValue or NumberValue")
        return
    end
    local amount = randomGen:NextInteger(CONFIG.CoinsMin, CONFIG.CoinsMax)
    coins.Value = coins.Value + amount
    notifyReward("+" .. amount .. " Lucky Coins!")
end

local function restoreParts()
    pose(originalPivot, 1, 0, 0)
    for _, data in ipairs(parts) do
        data.part.Transparency = data.transparency
        data.part.Color = data.color
        data.part.CanCollide = data.canCollide
        data.part.CanQuery = data.canQuery
    end
    light.Brightness = 0
end
local function hideParts()
    for _, data in ipairs(parts) do
        data.part.Transparency = 1
        data.part.CanCollide = false
        data.part.CanQuery = false
    end
end
local function clearEffects()
    if activeEffects then activeEffects:Destroy(); activeEffects = nil end
end

local opening = false
local alive = true
local connection
connection = prompt.Triggered:Connect(function(player)
    if opening or not alive or not player or not player:IsA("Player") or player.Parent ~= Players then return end
    local character = player.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not hrp or not humanoid or humanoid.Health <= 0 then return end
    if (hrp.Position - interaction.WorldPosition).Magnitude > activationDistance + 3 then return end
    opening = true
    prompt.Enabled = false
    -- Capture the current reference each activation; permits moving/rotating
    -- the whole model with PivotTo() between activations without losing the pose.
    originalPivot = root.CFrame
    cancelTweens()
    clearEffects()
    activeEffects = Instance.new("Folder")
    activeEffects.Name = "SkybrickOpeningFX"
    activeEffects.Parent = Workspace
    Debris:AddItem(activeEffects, CONFIG.OpenTime + 10)
    local orbs = {}

    local ok, err = pcall(function()
        local center = originalPivot:PointToWorldSpace(bodyCenter)
        for _, data in ipairs(parts) do data.part.CanCollide = false end
        emit(goldDust, 45)
        emit(skyDust, 25)
        scaleTween(light, 0.45, {Brightness = 3.8})
        for _, data in ipairs(parts) do
            if data.group == "Body" or data.group == "Lid" then
                scaleTween(data.part, 0.25, {Color = Color3.fromRGB(255,244,156)})
            end
        end
        shockwave(center, Color3.fromRGB(255,222,77), 0.7, 8)
        for i = 1, CONFIG.OrbitingStars do
            local orb = sphereEffect(0.13 + (i % 3) * 0.045,
                i % 3 == 0 and Color3.fromRGB(145,231,255) or Color3.fromRGB(255,239,85), 0)
            orb.Position = center
            local att0 = Instance.new("Attachment"); att0.Position = Vector3.new(0,0.08 * unit,0); att0.Parent = orb
            local att1 = Instance.new("Attachment"); att1.Position = Vector3.new(0,-0.08 * unit,0); att1.Parent = orb
            local trail = Instance.new("Trail")
            trail.Attachment0 = att0; trail.Attachment1 = att1
            trail.Lifetime = 0.21
            trail.LightEmission = 1
            trail.Transparency = NumberSequence.new(0.1, 1)
            trail.Color = ColorSequence.new(orb.Color)
            trail.Parent = orb
            table.insert(orbs, orb)
            Debris:AddItem(orb, CONFIG.OpenTime + 2.5)
        end

        local startTime = os.clock()
        local lastEmit, lastPulse = 0, 0
        while alive and os.clock() - startTime < CONFIG.OpenTime do
            RunService.Heartbeat:Wait()
            if not alive then return end
            local elapsed = os.clock() - startTime
            local t = math.clamp(elapsed / CONFIG.OpenTime, 0, 1)
            local charging = t * t
            local speed = 6 + 29 * charging
            local amplitude = (0.012 + 0.20 * charging) * unit
            local jitter = CFrame.new(randomGen:NextNumber(-amplitude, amplitude),
                randomGen:NextNumber(-amplitude, amplitude), randomGen:NextNumber(-amplitude, amplitude))
                * CFrame.Angles(math.rad(randomGen:NextNumber(-2,2) * charging),
                    math.rad(randomGen:NextNumber(-2,2) * charging),
                    math.rad(randomGen:NextNumber(-4,4) * charging))
            local wingFlap = math.sin(elapsed * speed) * math.rad(CONFIG.WingFlapAmount) * (0.30 + 0.70 * t)
            local pulse = 1 + 0.04 * math.sin(elapsed * (15 + 20 * charging)) + 0.12 * charging
            pose(originalPivot * jitter, pulse, wingFlap, 0)
            light.Brightness = 2.0 + 4.5 * t + math.abs(math.sin(elapsed * 18))
            for i, orb in ipairs(orbs) do
                local phi = (i / #orbs) * math.pi * 2 + elapsed * (2 + 8 * t) * (i % 2 == 0 and 1 or -1)
                local radius = 2.35 + 1.45 * math.sin(elapsed * 2 + i) * 0.25 + 0.7 * t
                local h = 1.15 * math.sin(phi * 1.8 + elapsed * 4 + i) + 1.2
                orb.Position = center + Vector3.new(math.cos(phi) * radius, h, math.sin(phi) * radius) * unit
            end
            if elapsed - lastEmit > 0.14 then
                emit(goldDust, 7 + 18 * t)
                emit(skyDust, 4 + 12 * t)
                lastEmit = elapsed
            end
            if elapsed - lastPulse > (0.70 - 0.36 * t) then
                shockwave(center, t > 0.5 and Color3.fromRGB(172,229,255) or Color3.fromRGB(255,221,76),
                    0.55, 7 + 5 * t)
                lastPulse = elapsed
            end
        end
        if not alive then return end

        -- Original HUGE POP effects, now with an independently flying lid.
        restoreParts()
        for _, data in ipairs(parts) do data.part.CanCollide = false end
        local blast = sphereEffect(0.1, Color3.fromRGB(255,250,200), 0.15)
        blast.CFrame = CFrame.new(center)
        scaleTween(blast, 0.38, {Size = Vector3.new(9,9,9) * unit, Transparency = 1}, Enum.EasingStyle.Exponential)
        Debris:AddItem(blast, 0.6)
        shockwave(center, Color3.fromRGB(255,224,67), 0.85, 15)
        shockwave(center, Color3.fromRGB(111,214,255), 1.1, 19)
        emit(goldDust, 130)
        emit(skyDust, 95)
        emit(burstDust, 170)
        scaleTween(light, 0.17, {Brightness = 14})
        -- Do not let a missing Backpack pause the pop / fading choreography.
        task.spawn(function()
            local rewarded, rewardError = pcall(giveReward, player)
            if not rewarded then warn("LuckyBlock reward error: " .. tostring(rewardError)) end
        end)
        for _, data in ipairs(parts) do
            scaleTween(data.part, 0.42, {Transparency = 1}, Enum.EasingStyle.Back)
        end
        for _, orb in ipairs(orbs) do
            scaleTween(orb, 0.38, {Size = Vector3.new(0.02,0.02,0.02) * unit, Transparency = 1})
            Debris:AddItem(orb, 1)
        end
        local popStart = os.clock()
        while alive and os.clock() - popStart < 0.5 do
            RunService.Heartbeat:Wait()
            if not alive then return end
            local alpha = math.clamp((os.clock() - popStart) / 0.42, 0, 1)
            local pop = TweenService:GetValue(alpha, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            pose(originalPivot, 1, math.rad(CONFIG.WingFlapAmount * 1.25) * pop, pop)
        end
        light.Brightness = 0
        hideParts()
    end)

    if not alive then return end
    if not ok then
        warn("LuckyBlock animation error: " .. tostring(err))
        cancelTweens()
        clearEffects()
        goldDust:Clear(); skyDust:Clear(); burstDust:Clear()
        restoreParts()
        prompt.Enabled = true
        opening = false
        return
    end
    for _, orb in ipairs(orbs) do if orb.Parent then Debris:AddItem(orb, 1) end end
    if CONFIG.RespawnSeconds > 0 then
        task.wait(CONFIG.RespawnSeconds)
        if not alive then return end
        cancelTweens()
        clearEffects()
        restoreParts()
        emit(goldDust, 20)
        prompt.Enabled = true
        opening = false
    else
        -- Keep ALL four groups invisible and non-collidable until removed.
        task.delay(3, function()
            if alive then cancelTweens(); clearEffects() end
        end)
    end
end)

    alive = false
    connection:Disconnect()
    cancelTweens()
    clearEffects()
    if luckyBlock.Parent then restoreParts() end
    interaction:Destroy()
    attachment:Destroy()
    controller:Destroy()
end)
prompt.Enabled = true
luckyBlock:SetAttribute("SkybrickStatus", "Ready")
print(string.format("[Skybrick] READY: %s | %d parts | E distance %.1f studs",
    luckyBlock:GetFullName(), #parts, activationDistance))
if not ProximityPromptService.Enabled then
    warn("[Skybrick] ProximityPromptService.Enabled is false: Roblox has hidden ALL prompts. Enable it in Explorer or check the script that disables it")
end
end -- start

if not ok then
    if cleanupStartup then cleanupStartup() end
    if script.Parent then script.Parent:SetAttribute("SkybrickStatus", "Error: " .. tostring(message)) end
    warn("[Skybrick] STARTUP ERROR: " .. tostring(message))
end

-- Disable only recognized animation controllers, keeping unrelated scripts.
local disabled = 0
for _, item in ipairs(model:GetDescendants()) do
    if item:IsA("BaseScript") then
        local readable, text = pcall(function() return item.Source end)
        local recognized = readable and (string.find(text, "SKYBRICK", 1, true)
            or string.find(text, "WINGED LUCKY BLOCK", 1, true))
        if recognized or item == selected then
            item.Enabled = false
            disabled = disabled + 1
        elseif item.Enabled then
            warn("[Skybrick] Another script remains enabled: " .. item:GetFullName()
                .. ". Check that it is not another opening animation")
        end
    end
end
local opening = model:FindFirstChild("Opening_Skybrick")
if opening and not opening:IsA("Script") then
    opening.Name = "Opening_Skybrick_Old"
    opening = nil
end
if not opening then
    opening = Instance.new("Script")
    opening.Name = "Opening_Skybrick"
    opening.Parent = model
end
opening.Enabled = false
opening.Source = source
opening.RunContext = Enum.RunContext.Server
if not model:IsDescendantOf(workspace) then model.Parent = workspace end
opening.Enabled = true
model:SetAttribute("SkybrickStatus", "Installed: press Play")
game:GetService("Selection"):Set({opening})
print("[Skybrick] INSTALLED: " .. opening:GetFullName() .. "; disabled old controllers: " .. disabled)
print("[Skybrick] Press PLAY (F5), approach the block and hold E. Output should show [Skybrick] READY")
if not game:GetService("ProximityPromptService").Enabled then
    warn("[Skybrick] ProximityPromptService.Enabled is false; enable it to display prompts")
end
